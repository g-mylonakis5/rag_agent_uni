def calculate_efficiency():
    # Load files dynamically
    files = glob.glob('data/box_scores/*.csv')
    if not files:
        print("No box score files found in the directory.")
        return

    df_list = [pd.read_csv(f) for f in files]
    df = pd.concat(df_list, ignore_index=True)

    # Convert MIN column
    df['MIN'] = df['MIN'].astype(str).apply(
        lambda x: float(x.split(':')[0]) + float(x.split(':')[1])/60.0 
        if ':' in x else (float(x) if x.replace('.','',1).isdigit() else 0.0)
    )

    # Filter for Dorsey
    dorsey_df = df[df['Player'].str.contains('dorsey', case=False, na=False)].copy()

    if dorsey_df.empty:
        print("No data found for player Dorsey.")
        return

    # Helper to parse shooting splits
    def parse_split(col_name):
        split_data = dorsey_df[col_name].astype(str).str.replace('-', '/').str.split('/', expand=True).iloc[:, :2].fillna(0).astype(int)
        return split_data[0], split_data[1]

    f2m, f2a = parse_split('2FG')
    f3m, f3a = parse_split('3FG')
    ftm, fta = parse_split('FT')

    # Aggregate totals
    total_pts = dorsey_df['PTS'].sum()
    total_f2a = f2a.sum()
    total_f3a = f3a.sum()
    total_fta = fta.sum()
    total_f2m = f2m.sum()
    total_f3m = f3m.sum()
    total_ftm = ftm.sum()

    # Calculate eFG% = (FGM + 0.5 * 3PM) / FGA
    total_fga = total_f2a + total_f3a
    if total_fga > 0:
        efg = (total_f2m + total_f3m + 0.5 * total_f3m) / total_fga
    else:
        efg = 0.0

    # Calculate TS% = PTS / (2 * (FGA + 0.44 * FTA))
    ts_denom = 2 * (total_fga + 0.44 * total_fta)
    if ts_denom > 0:
        ts = total_pts / ts_denom
    else:
        ts = 0.0

    print(f"Tyler Dorsey Efficiency Metrics:")
    print(f"Effective Field Goal Percentage (eFG%): {efg:.3f}")
    print(f"True Shooting Percentage (TS%): {ts:.3f}")

    if ts > efg:
        print("True Shooting Percentage is higher than Effective Field Goal Percentage.")
    elif efg > ts:
        print("Effective Field Goal Percentage is higher than True Shooting Percentage.")
    else:
        print("Both efficiency metrics are equal.")
